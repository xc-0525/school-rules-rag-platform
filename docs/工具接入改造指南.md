# 校园服务只读工具接入指南

> 目标：把已实现的报修、预约、失物招领服务接入问答 Agent 的 Tool Calling，使"我报的修到哪了""明天图书馆还有位置吗""有人捡到校园卡吗"这类**实时业务问题**由模型自主调工具回答，而不是硬编进提示词。
>
> 原则：**只读、按用户隔离、失败降级**。写操作一律留在页面，模型不碰。
> 改动范围：4 个文件 + 1 个测试文件，不动核心链路（图引擎、检索、生成）。

```python
def active_tool_definitions(settings: Any) -> list[dict[str, Any]]:
    """按开关返回可用工具，供“有/无工具”消融实验复用同一份代码。"""
    if getattr(settings, "campus_tools_enabled", True):
        return TOOL_DEFINITIONS
```

```python
    # ---------------------------------------------------------- 校园服务只读工具
    async def _query_repair_status(
        self, arguments: dict[str, Any], user_id: int
    ) -> ToolResult:
        from app.services import repairs as repair_service

        status = str(arguments.get("status") or "").strip()
        limit = max(1, min(int(arguments.get("limit") or 5), 10))
        try:
            rows = await asyncio.to_thread(
                lambda: repair_service.list_repairs(
                    self.settings,
                    user_id=user_id,
                    is_admin=False,      # 只查本人，越权面为零
                    status=status,
                    limit=limit,
                )
            )
        except Exception:  # noqa: BLE001 —— 表未初始化时降级为空，不中断问答
            logger.exception("query_repair_status failed")
            rows = []

        if not rows:
            summary = "没有查到你的报修记录。" if not status else f"没有处于 {status} 状态的报修记录。"
            return ToolResult(
                name="query_repair_status",
                arguments={"status": status, "limit": limit},
                success=True,
                summary=summary,
                payload={"repairs": []},
            )

        labels = {"pending": "待受理", "processing": "处理中",
                  "completed": "已完成", "closed": "已关闭"}
        lines = []
        for index, row in enumerate(rows, 1):
            line = (
                f"{index}. {row.get('repair_no') or row.get('id')}"
                f"｜{row.get('title', '')}"
                f"｜{labels.get(row.get('status', ''), row.get('status', ''))}"
                f"｜{row.get('building', '')}{row.get('room', '')}"
                f"｜{row.get('created_at', '')}"
            )
            if row.get("handler_remark"):
                line += f"\n   处理意见：{row['handler_remark']}"
            lines.append(line)

        return ToolResult(
            name="query_repair_status",
            arguments={"status": status, "limit": limit},
            success=True,
            summary="你最近的报修记录：\n\n" + "\n".join(lines),
            sources=[
                SourceItem(
                    type="tool",
                    title=f"报修 {row.get('repair_no') or row.get('id')}",
                    detail=labels.get(row.get("status", ""), row.get("status", "")),
                    snippet=f"{row.get('title', '')} {row.get('description', '')}"[:200],
                )
                for row in rows
            ],
            payload={"repairs": rows},
        )
```

```python
    async def _query_location_availability(
        self, arguments: dict[str, Any]
    ) -> ToolResult:
        from app.services import reservations as reservation_service

        keyword = str(arguments.get("location") or "").strip()
        res_date = str(arguments.get("date") or "").strip() or (
            datetime.now().date() + timedelta(days=1)
        ).isoformat()

        locations = await asyncio.to_thread(
            lambda: reservation_service.list_locations(self.settings)
        )
        if keyword:
            locations = [loc for loc in locations if keyword in str(loc.get("name", ""))]
        if not locations:
            return ToolResult(
                name="query_location_availability",
                arguments={"location": keyword, "date": res_date},
                success=True,
                summary=f"没有找到名称包含“{keyword}”的可预约位置。",
                payload={"locations": []},
            )

        lines, payload = [], []
        for loc in locations[:3]:
            try:
                data = await asyncio.to_thread(
                    lambda loc=loc: reservation_service.availability(
                        self.settings, location_id=loc["id"], res_date=res_date
                    )
                )
            except Exception:  # noqa: BLE001 —— 日期越界等规则错误不拖垮整次调用
                logger.exception("availability failed: %s", loc.get("name"))
                continue
            free = [slot["label"] for slot in data["slots"] if not slot["booked"]]
            head = f"{loc.get('name', '')}（{loc.get('zone', '')}）{res_date}"
            if data.get("range_note"):
                lines.append(f"{head}：{data['range_note']}")
            else:
                lines.append(f"{head}：可预约 {('、'.join(free) or '无')}")
            payload.append({"location": loc.get("name"), "date": res_date,
                            "free_slots": free, "range_note": data.get("range_note", "")})

        return ToolResult(
            name="query_location_availability",
            arguments={"location": keyword, "date": res_date},
            success=True,
            summary="场馆余量查询结果：\n\n" + "\n".join(lines),
            sources=[
                SourceItem(
                    type="tool",
                    title=f"余量：{item['location']}",
                    detail=res_date,
                    snippet=("可预约 " + "、".join(item["free_slots"]))
                    if item["free_slots"] else "该日无可预约时段",
                )
                for item in payload
            ],
            payload={"locations": payload},
        )
```

```python
    async def _query_lost_found(
        self, arguments: dict[str, Any], user_id: int
    ) -> ToolResult:
        from app.services import lostfound as lostfound_service

        keyword = str(arguments.get("keyword") or "").strip()
        kind = str(arguments.get("kind") or "").strip()
        limit = max(1, min(int(arguments.get("limit") or 5), 10))
        try:
            rows = await asyncio.to_thread(
                lambda: lostfound_service.list_items(
                    self.settings, user_id=user_id, is_admin=False,
                    kind=kind, scope="public", limit=min(limit * 4, 50),
                )
            )
        except Exception:  # noqa: BLE001
            logger.exception("query_lost_found failed")
            rows = []

        if keyword:  # list_items 无 keyword 参数，在结果内做一次包含匹配
            rows = [
                row for row in rows
                if keyword in f"{row.get('title', '')}{row.get('description', '')}{row.get('location', '')}"
            ]
        rows = rows[:limit]

        if not rows:
            summary = "失物招领公示栏暂未查到相关信息。"
            return ToolResult(name="query_lost_found",
                              arguments={"keyword": keyword, "kind": kind, "limit": limit},
                              success=True, summary=summary, payload={"items": []})

        lines = [
            f"{index}. 【{row.get('kind_label', '')}】{row.get('title', '')}"
            f"｜地点：{row.get('location', '')}｜{row.get('created_at', '')}"
            f"\n   {row.get('description', '')[:120]}"
            for index, row in enumerate(rows, 1)
        ]
        return ToolResult(
            name="query_lost_found",
            arguments={"keyword": keyword, "kind": kind, "limit": limit},
            success=True,
            summary="失物招领公示栏检索结果：\n\n" + "\n".join(lines),
            sources=[
                SourceItem(type="tool", title=row.get("title", "失物信息"),
                           detail=row.get("kind_label", ""),
                           snippet=row.get("description", "")[:200])
                for row in rows
            ],
            payload={"items": rows},
        )
```

> 说明：`sources` 与 `payload` 同源同序，前端"来源引用"直接按 `title` 展示，不需要改前端组件。

## 4. 一个必须先修的坑：`TICKET_INTENT_KEYWORDS`

`tools.py` 第 28 行附近的关键词表包含 **"报修"、"失物"**。这会导致用户问"我的报修到哪了"时命中 `_has_ticket_intent`，模型可能直接调 `create_ticket` 去建单——**明明是查询，却写了一条工单**。

改法：把"报修""失物"从该表移除，只保留动作性表述（投诉/工单/转人工/反馈/举报/维修申请等），或改为"报修申请""我要报修"这类短语。

```python
TICKET_INTENT_KEYWORDS = (
    "投诉", "工单", "转人工", "人工", "反馈", "举报", "建议",
    "故障", "坏了", "噪音", "扰民", "太吵", "维修", "挂失",
    "我要报修", "申请报修", "提交报修",
)
```

## 5. `app/agents/prompts.py`：告诉模型什么时候用

在系统提示词的"工具使用"段补 3 句（不要写长，写长会抢走 RAG 的调用）：

```
- 涉及“我自己的”报修进度、预约记录时，调用 query_repair_status / query_my_reservations，不要用知识库推测。
- 涉及“某天某场馆还有没有位置”时，调用 query_location_availability。
- 涉及“有没有人捡到/丢了某物”时，调用 query_lost_found；发布失物请引导用户到失物招领页面，不要调用 create_ticket。
- 工具返回“查无记录”时，如实告知并给出下一步（去页面提交/联系管理员），不要编造单号或时段。
```

## 9. 不要做的事

| 想法 | 为什么不做 |
| --- | --- |
| 让模型调 `create_repair` / `create_reservation` 直接下单 | 模型会把"我想预约"理解成"已预约"，产生脏数据；写入必须走页面 + 用户确认 |
| 给工具加 `is_admin=True` 让管理员查全部 | 工具入参只有 `user_id`，没有角色；越权一旦开口就收不住。需要管理视角就走管理端页面 |
| 在工具里返回完整 `contact`（手机号） | 失物招领的联系方式应打码，工具摘要里只留前 3 位 + `****` |
| 用 `search_rules` 兜这两类问题 | 报修进度、预约余量是**库里的实时数据**，不在语料里；用 RAG 猜一定答错，还会污染检索评测 |
